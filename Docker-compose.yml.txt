version: '3'
services:
  db:
    image: postgres:14
    volumes:
      - pgdata:/var/lib/postgresql/data
    environment:
      POSTGRES_PASSWORD: password
      POSTGRES_USER: user
      POSTGRES_DB: myapp_development
  web:
    build: .

    volumes:
      - .:/app
    ports:
      - "3000:3000"
    depends_on:
      - db

  pgdata
